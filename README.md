
# MY FIRST FINGER COUNTER PROGRAM

print("WELCOME TO AI FINGER COUNTER")

fingers = int(input("Enter the number of fingers: "))

if fingers == 1:
    print("ONE")

elif fingers == 2:
    print("TWO")

elif fingers == 3:
    print("THREE")

elif fingers == 4:
    print("FOUR")

elif fingers == 5:
    print("FIVE")

else:
    print("Please enter a number from 1 to 5")
